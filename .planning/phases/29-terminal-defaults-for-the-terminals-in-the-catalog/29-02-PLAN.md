The probe asks kitty for each chord's definitions with kitty's own `parse_shortcut` (`kitty_mod` is the default ctrl+shift in the generated files); kitty keeps every definition of a chord in order and the last one wins, so the check compares the last definition:

```python
import json, shutil, subprocess, sys, tempfile
from pathlib import Path
from installer.terminal_keys import TERMINAL_KEYS, apply_terminal_keys

EXPECTED = {
    "fedora": {
        "ctrl+shift+k": "send_text all \\x0c",
        "kitty_mod+t": "new_tab_with_cwd",
        "kitty_mod+enter": "new_window_with_cwd",
        "kitty_mod+n": "new_os_window_with_cwd",
    },
    "macos": {
        "opt+left": "send_text all \\x1bb",
        "opt+right": "send_text all \\x1bf",
        "cmd+k": "send_text all \\x0c",
        "cmd+t": "new_tab_with_cwd",
        "cmd+enter": "new_window_with_cwd",
        "cmd+n": "new_os_window_with_cwd",
    },
}
PROBE = (
    "import json, sys\n"
    "from kitty.config import load_config\n"
    "from kitty.options.utils import parse_shortcut\n"
    "opts = load_config(sys.argv[-2])\n"
    "keymap = opts.keyboard_modes[''].keymap\n"
    "maps = {c: [d.definition for d in keymap.get(parse_shortcut(c.replace('kitty_mod', 'ctrl+shift')), [])]"
    " for c in json.loads(sys.argv[-1])}\n"
    "print(json.dumps({'maps': maps, 'macos_option_as_alt': opts.macos_option_as_alt}))\n"
)

kitty = shutil.which("kitty")
if kitty is None:
    sys.exit("not run: no kitty on PATH")
root = Path(tempfile.mkdtemp(prefix="kitty-check-"))
problems = []
for platform_os, expected in EXPECTED.items():
    home = root / platform_os
    env = {"HOME": str(home)}
    lines, refused = apply_terminal_keys(TERMINAL_KEYS["kitty"], "kitty", platform_os, home, env, skip_recorded=False)
    config = TERMINAL_KEYS["kitty"].config_path(platform_os, home, env)
    print(platform_os, refused, lines)
    if refused is not None:
        problems.append(f"{platform_os}: the row refused: {refused}")
        continue
    run_env = {"HOME": str(home), "XDG_CONFIG_HOME": str(home / "c"), "XDG_CACHE_HOME": str(home / "cache"), "PATH": "/usr/bin:/bin"}
    out = subprocess.run([kitty, "+runpy", PROBE, str(config), json.dumps(list(expected))], env=run_env, capture_output=True, text=True, timeout=60)
    if out.returncode != 0:
        problems.append(f"{platform_os}: kitty exited {out.returncode}: {out.stderr.strip()[-500:]}")
        continue
    try:
        result = json.loads(out.stdout.strip().splitlines()[-1])
    except (IndexError, ValueError):
        problems.append(f"{platform_os}: no parseable probe output: {out.stdout[-500:]!r}")
        continue
    for chord, action in expected.items():
        found = result["maps"].get(chord) or ["<no definition>"]
        if found[-1] != action:
            problems.append(f"{platform_os}: {chord} is {found[-1]!r}, expected {action!r}")
    if result["macos_option_as_alt"] != 0:
        problems.append(f"{platform_os}: macos_option_as_alt is {result['macos_option_as_alt']!r}, expected 0")
shutil.rmtree(root, ignore_errors=True)
if problems:
    print("\n".join(problems))
    sys.exit(1)
print("KITTY_LOAD_OK")
```

Expected: one `<os> None (<lines>)` line per OS, then `KITTY_LOAD_OK` as the last line, exit status 0. Any mismatch line or a non-zero exit is a finding. With no kitty on PATH it prints `not run: no kitty on PATH` and exits 1 (no `KITTY_LOAD_OK`, so it is never mistaken for a pass).
